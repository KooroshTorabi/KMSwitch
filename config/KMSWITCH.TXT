[general]

edge_corner_block = on    # no switch while the pointer is in a screen corner

auto_measure = on         # measure a newly connected PC by itself
config_ui = on            # allow the open_config and measure hotkeys to type into the PC

measure = WIN+F9          # measure the screens of both PCs (LEDs blink)
